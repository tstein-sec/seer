### HIGH ROI — secrets, logs, backups
# AWS keys pasted into public GitLab repos/issues (classic leak)
inurl:gitlab "AWS_SECRET_KEY" [3]
# Runner logs can leak tokens/endpoints
filetype:log "GitLab Runner" [1]
# sshd auth log exposure on GitLab hosts

# Instance backup archives left in web root
inurl:gitlab-backup [1]
